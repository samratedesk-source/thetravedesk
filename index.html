function doGet(e) { try { var action = e.parameter.action; var callback = e.parameter.callback; var fileName = "TheTravelDesk_AutoBackup.json"; var result = {}; if (action == "load") { var files = DriveApp.getFilesByName(fileName); if (!files.hasNext()) { result = {error: "No backup found"}; } else { var file = files.next(); var content = file.getBlob().getDataAsString(); try { result = JSON.parse(content); } catch { result = {error: "Invalid backup format", raw: content.substring(0,500)}; } } } else if (action == "save") { var data = e.parameter.data; if (!data) { result = {error: "No data"}; } else { saveFile(fileName, data); result = {success: true, timestamp: new Date().toISOString(), size: data.length}; } } else { result = {error: "Invalid action"}; } var json = JSON.stringify(result); // JSONP Support for CORS fix if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } catch(err) { var errResult = JSON.stringify({error: err.toString()}); if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + errResult + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(errResult) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var fileName = "TheTravelDesk_AutoBackup.json"; var data = ""; if (e.postData && e.postData.contents) { var contents = e.postData.contents; try { var json = JSON.parse(contents); if (json.data) data = json.data; else if (json.localStorage) data = JSON.stringify(json); else data = contents; } catch { data = contents; } } else if (e.parameter.data) { data = e.parameter.data; } if (!data) { return ContentService.createTextOutput(JSON.stringify({error: "No data in POST"})) .setMimeType(ContentService.MimeType.JSON); } saveFile(fileName, data); return ContentService.createTextOutput(JSON.stringify({success: true, timestamp: new Date().toISOString(), size: data.length})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function saveFile(fileName, data) { var files = DriveApp.getFilesByName(fileName); if (files.hasNext()) { var file = files.next(); file.setContent(data); } else { DriveApp.createFile(fileName, data, MimeType.PLAIN_TEXT); } }
